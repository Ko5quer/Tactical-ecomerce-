@api_view(["GET"])
def get_product(request, product_id):
    try:
        product = Product.objects.get(id=product_id)
    except Product.DoesNotExist:
        return Response(
            {"error": "Product not found"},
            status=status.HTTP_404_NOT_FOUND,
        )
    serializer = ProductSerializer(product, context={"request": request})
    return Response(serializer.data)


@api_view(["GET"])
def featured_products(request):
    """Random in-stock products, spread across different categories.

    GET /api/products/featured/?limit=8
    """
    try:
        limit = max(1, min(int(request.query_params.get("limit", 8)), 24))
    except ValueError:
        limit = 8

    # (product id, category id) for everything in stock, shuffled
    rows = list(
        Product.objects.filter(current_stock__gt=0).values_list("id", "category_id")
    )
    random.shuffle(rows)

    buckets = defaultdict(list)
    for product_id, category_id in rows:
        buckets[category_id].append(product_id)

    categories = list(buckets)
    random.shuffle(categories)

    # take one product from each category in turn until we have enough
    picked = []
    while len(picked) < limit and any(buckets.values()):
        for category_id in categories:
            if buckets[category_id] and len(picked) < limit:
                picked.append(buckets[category_id].pop())

    products = {p.id: p for p in Product.objects.filter(id__in=picked)}
    ordered = [products[i] for i in picked]

    serializer = ProductSerializer(ordered, many=True, context={"request": request})
    return Response(serializer.data)


@api_view(["GET"])
def search_products(request):
    """Search products (name, description, category name) and categories (name).

    GET /api/products/search/?q=cargo&limit=24
    Every word typed must match somewhere, so "black cargo" narrows the results.
    """
    q = " ".join((request.query_params.get("q") or "").split())[:100]

    if len(q) < 2:
        return Response({"query": q, "products": [], "categories": []})

    try:
        limit = max(1, min(int(request.query_params.get("limit", 24)), 48))
    except ValueError:
        limit = 24

    products = Product.objects.select_related("category")
    for term in q.split()[:6]:
        products = products.filter(
            Q(product_name__icontains=term)
            | Q(description__icontains=term)
            | Q(category__name__icontains=term)
        )
    products = products.order_by("product_name")[:limit]

    categories = (
        Category.objects.filter(name__icontains=q)
        .prefetch_related("products")
        .order_by("name")[:8]
    )

    context = {"request": request}
    return Response(
        {
            "query": q,
            "products": ProductSerializer(products, many=True, context=context).data,
            "categories": CategorySerializer(categories, many=True, context=context).data,
        }
    )

        self.client = APIClient()

        self.category = Category.objects.create(
            name="Test Category",
        )

        self.product = Product.objects.create(
            category=self.category,
            product_name="Test Product",
            description="A product created for testing",
            product_price=Decimal("99.99"),
            current_stock=10,
        )

    def test_get_existing_product(self):
        url = reverse("get_product", args=[self.product.id])

        response = self.client.get(url)

        self.assertEqual(
            response.status_code,
            status.HTTP_200_OK,
        )

        expected_data = ProductSerializer(self.product).data
        self.assertEqual(response.data, expected_data)

    def test_get_nonexistent_product(self):
        url = reverse("get_product", args=[999999])

        response = self.client.get(url)

        self.assertEqual(
            response.status_code,
            status.HTTP_404_NOT_FOUND,
        )
        self.assertEqual(
            response.data,
            {"error": "Product not found"},
        )

    def test_get_product_returns_correct_name(self):
        url = reverse("get_product", args=[self.product.id])

        response = self.client.get(url)

        self.assertEqual(
            response.data["product_name"],
            "Test Product",
        )

    def test_get_product_returns_correct_price(self):
        url = reverse("get_product", args=[self.product.id])

        response = self.client.get(url)

        self.assertEqual(
            Decimal(str(response.data["product_price"])),
            Decimal("99.99"),
        )

    def test_get_product_returns_correct_stock(self):
        url = reverse("get_product", args=[self.product.id])

        response = self.client.get(url)

        self.assertEqual(response.data["current_stock"], 10)

            status.HTTP_405_METHOD_NOT_ALLOWED,
        )

        self.product.refresh_from_db()
        self.assertEqual(
            self.product.product_name,
            "Test Product",
        )

class Product(models.Model):

        "categories.Category",
        on_delete=models.CASCADE,
        related_name="products",
    )
    product_name = models.CharField(max_length=255)
    description = models.TextField()
    product_price = models.DecimalField(
        max_digits=10,
        decimal_places=2,
    )
    current_stock = models.PositiveIntegerField(default=0)
    image = models.ImageField(
        upload_to="products/",
        null=True,
        blank=True,
    )

    class Meta:
        db_table = "products"
        constraints = [
            models.CheckConstraint(
                condition=models.Q(current_stock__gte=0),
                name="product_stock_nonnegative",
            ),
        ]

    def __str__(self):
        return self.product_name

import re

from django.contrib.auth.password_validation import validate_password
from rest_framework import serializers

from .models import User

NAME_RE = re.compile(r"^[^\W\d_](?:[^\W\d_]|[ '.\-])*$")


class RegisterSerializer(serializers.ModelSerializer):
    password = serializers.CharField(write_only=True, max_length=128)

    class Meta:
        model = User
        fields = ["user_id", "email", "first_name", "last_name", "password"]
        read_only_fields = ["user_id"]
        extra_kwargs = {
            "first_name": {"required": True, "allow_blank": False},
            "last_name": {"required": False, "allow_blank": True},
        }

    def _check_name(self, value):
        if value and not NAME_RE.match(value):
            raise serializers.ValidationError(
                "Only letters, spaces, apostrophes, full stops and hyphens are allowed."
            )
        return value

    def validate_first_name(self, value):
        return self._check_name(value)

    def validate_last_name(self, value):
        return self._check_name(value)

    def validate_email(self, value):
        value = value.lower()
        if User.objects.filter(email__iexact=value).exists():
            raise serializers.ValidationError("An account with this email already exists.")
        return value

    def validate_password(self, value):
        validate_password(value)
        return value

    def create(self, validated_data):
        return User.objects.create_user(**validated_data)


class UserSerializer(serializers.ModelSerializer):
    class Meta:
        model = User
        fields = ["user_id", "email", "first_name", "last_name"]

from django.contrib.auth.models import (
    AbstractBaseUser,
    BaseUserManager,
    PermissionsMixin,
)
from django.db import models
from django.utils import timezone


class UserManager(BaseUserManager):
    use_in_migrations = True

    def create_user(self, email, password=None, **extra_fields):
        if not email:
            raise ValueError("Email is required")
        email = self.normalize_email(email).lower()
        user = self.model(email=email, **extra_fields)
        user.set_password(password)
        user.save(using=self._db)
        return user

    def create_superuser(self, email, password=None, **extra_fields):
        extra_fields.setdefault("is_staff", True)
        extra_fields.setdefault("is_superuser", True)
        return self.create_user(email, password, **extra_fields)


class User(AbstractBaseUser, PermissionsMixin):
    user_id = models.AutoField(primary_key=True)
    email = models.EmailField(unique=True)
    first_name = models.CharField(max_length=150, blank=True)
    last_name = models.CharField(max_length=150, blank=True)

    is_active = models.BooleanField(default=True)
    is_staff = models.BooleanField(default=False)
    date_joined = models.DateTimeField(default=timezone.now)

    USERNAME_FIELD = "email"
    REQUIRED_FIELDS = []

    objects = UserManager()

    class Meta:
        db_table = "users"

    def __str__(self):
        return self.email

def _bad_request(message):
    return Response({"error": message}, status=status.HTTP_400_BAD_REQUEST)


def _parse_quantity(raw, default=None):
    """Returns a positive int or None if invalid."""
    if raw is None:
        raw = default
    if raw is None or isinstance(raw, bool):
        return None
    try:
        value = int(raw)
    except (TypeError, ValueError):
        return None
    return value if value >= 1 else None


@api_view(["GET"])
@permission_classes([IsAuthenticated])
def get_cart(request):
    cart, created = Cart.objects.get_or_create(user=request.user)
    serializer = CartSerializer(cart, context={"request": request})
    return Response(serializer.data, status=status.HTTP_200_OK)


@api_view(["POST"])
@permission_classes([IsAuthenticated])
@transaction.atomic
def add_cartItem(request):
    product_id = request.data.get("product_id")
    if product_id is None or isinstance(product_id, bool):
        return _bad_request("product_id is required")
    try:
        product_id = int(product_id)
    except (TypeError, ValueError):
        return _bad_request("product_id must be a number")

    quantity = _parse_quantity(request.data.get("quantity"), default=1)
    if quantity is None:
        return _bad_request("quantity must be a positive whole number")

    product = get_object_or_404(Product, pk=product_id)

    if product.current_stock < 1:
        return _bad_request(f"{product.product_name} is out of stock.")

    cart, created = Cart.objects.get_or_create(user=request.user)

    item = CartItem.objects.filter(cart=cart, product=product).first()
    already_in_cart = item.quantity if item else 0

    if already_in_cart + quantity > product.current_stock:
        left = product.current_stock - already_in_cart
        if left <= 0:
            return _bad_request(
                f"You already have all {product.current_stock} of {product.product_name} in your cart."
            )
        return _bad_request(
            f"Only {product.current_stock} of {product.product_name} in stock "
            f"(you can add {left} more)."
        )

    if item:
        item.quantity += quantity
        item.save(update_fields=["quantity", "updated_at"])
    else:
        item = CartItem.objects.create(cart=cart, product=product, quantity=quantity)

    return Response(
        {
            "message": "Product added to cart",
            "product_id": product.pk,
            "quantity": item.quantity,
        },
        status=status.HTTP_200_OK,
    )


@api_view(["PATCH"])
@permission_classes([IsAuthenticated])
def cartItem_quantity(request, item_id):
    quantity = _parse_quantity(request.data.get("quantity"))
    if quantity is None:
        return _bad_request("quantity must be a positive whole number")

    cart = get_object_or_404(Cart, user=request.user)
    item = get_object_or_404(
        CartItem.objects.select_related("product"), pk=item_id, cart=cart
    )

    if quantity > item.product.current_stock:
        return _bad_request(
            f"Only {item.product.current_stock} of {item.product.product_name} in stock."
        )

    item.quantity = quantity
    item.save(update_fields=["quantity", "updated_at"])

    return Response(
        {
            "message": "Cart item quantity updated",
            "item_id": item.id,
            "quantity": item.quantity,
        },
        status=status.HTTP_200_OK,
    )


@api_view(["DELETE"])

@api_view(["POST"])
@permission_classes([IsAuthenticated])
@transaction.atomic
def checkout(request):
    # 1. validate delivery details + payment method from the checkout page
    details = CheckoutSerializer(data=request.data)
    details.is_valid(raise_exception=True)
    info = details.validated_data
    payment_method = info.pop("payment_method")

    # 2. load the cart
    cart = get_object_or_404(
        Cart.objects.select_for_update(),
        user=request.user,
    )
    cart_items = list(cart.cart_items.select_related("product").all())

    if not cart_items:
        return Response(
            {"error": "Your cart is empty."},
            status=status.HTTP_400_BAD_REQUEST,
        )

    # 3. lock the products and check stock
    product_ids = sorted({item.product_id for item in cart_items})
    products = (
        Product.objects.select_for_update()
        .filter(pk__in=product_ids)
        .order_by("pk")
    )
    products_by_id = {product.pk: product for product in products}

    for item in cart_items:
        product = products_by_id[item.product_id]
        if product.current_stock < item.quantity:
            return Response(
                {"error": f"Insufficient stock for {product.product_name}."},
                status=status.HTTP_400_BAD_REQUEST,
            )

    # 4. work out the total on the server (never trust the browser's number)
    total = sum(
        (
            products_by_id[item.product_id].product_price * item.quantity
            for item in cart_items
        ),
        Decimal("0.00"),
    )

    # 5. create the order
    order = Order.objects.create(
        user=request.user,
        total=total,
        status="pending",
        payment_method=payment_method,
        payment_status="unpaid",
        **info,  # full_name, phone, address, city, postal_code, notes
    )

    for item in cart_items:
        product = products_by_id[item.product_id]

        OrderItem.objects.create(
            order=order,
            product=product,
            quantity=item.quantity,
            price=product.product_price,
        )

        product.current_stock -= item.quantity
        product.save(update_fields=["current_stock"])

    # 6. empty the cart only after the order items exist
    cart.cart_items.all().delete()

    return Response(OrderSerializer(order).data, status=status.HTTP_201_CREATED)

@permission_classes([IsAuthenticated])
def payment_instructions(request, order_id):
    """Private: how to pay for one of YOUR orders.

    Bank details are only returned to the customer who owns an unpaid EFT order,
    never to the public."""
    order = get_object_or_404(Order, pk=order_id, user=request.user)
    method = PaymentMethod.objects.filter(code=order.payment_method).first()

    data = {
        "method": order.payment_method,
        "label": method.label if method else order.get_payment_method_display(),
        "reference": order.reference,
        "amount": str(order.total),
        "payment_status": order.payment_status,
        "instructions": method.instructions if method else "",
        "bank_account": None,
    }

    if (
        order.payment_method == "eft"
        and order.payment_status != "paid"
        and order.status != "cancelled"
    ):
        account = BankAccount.objects.filter(is_active=True).first()
        if account:
            data["bank_account"] = BankAccountSerializer(account).data

    return Response(data)

# letters (any language), spaces, apostrophes, full stops and hyphens; must start with a letter
NAME_RE = re.compile(r"^[^\W\d_](?:[^\W\d_]|[ '.\-])*$")
PHONE_RE = re.compile(r"^\+?[0-9 ()\-]+$")
POSTAL_RE = re.compile(r"^[A-Za-z0-9 \-]{3,10}$")


def _has_control_chars(value, allow_newlines=False):
    allowed = {"\n", "\r"} if allow_newlines else set()
    return any(ord(c) < 32 and c not in allowed for c in value)


class CheckoutSerializer(serializers.Serializer):
    """Validates what the checkout page sends. Every message here is shown
    to the customer, so they are written in plain English."""

    full_name = serializers.CharField(max_length=150)
    phone = serializers.CharField(max_length=30)
    address = serializers.CharField(max_length=255)
    city = serializers.CharField(max_length=100)
    postal_code = serializers.CharField(
        max_length=20, required=False, allow_blank=True, default=""
    )
    notes = serializers.CharField(
        max_length=500, required=False, allow_blank=True, default=""
    )
    payment_method = serializers.CharField(max_length=20)

    def validate_full_name(self, value):
        if len(value) < 2:
            raise serializers.ValidationError("Enter your full name.")
        if not NAME_RE.match(value):
            raise serializers.ValidationError(
                "Name can only contain letters, spaces, apostrophes, full stops and hyphens."
            )
        return value

    def validate_phone(self, value):
        digits = re.sub(r"\D", "", value)
        if not PHONE_RE.match(value) or not 9 <= len(digits) <= 15:
            raise serializers.ValidationError(
                "Enter a valid phone number (9 to 15 digits, for example 082 123 4567)."
            )
        return value

    def validate_address(self, value):
        if len(value) < 5:
            raise serializers.ValidationError("Enter your full street address.")
        if not re.search(r"[^\W\d_]", value):
            raise serializers.ValidationError("The address must contain a street or place name.")
        if _has_control_chars(value):
            raise serializers.ValidationError("The address contains invalid characters.")
        return value

    def validate_city(self, value):
        if not NAME_RE.match(value):
            raise serializers.ValidationError("Enter a valid city or town name.")
        return value

    def validate_postal_code(self, value):
        if value and not POSTAL_RE.match(value):
            raise serializers.ValidationError(
                "Enter a valid postal code (3 to 10 letters, numbers, spaces or hyphens)."
            )
        return value

    def validate_notes(self, value):
        if _has_control_chars(value, allow_newlines=True):
            raise serializers.ValidationError("The notes contain invalid characters.")
        return value

    def validate_payment_method(self, value):
        ensure_default_payment_methods()
        if not PaymentMethod.objects.filter(code=value, is_enabled=True).exists():
            raise serializers.ValidationError("That payment method isn't available.")
        return value

from django.conf import settings
from django.db import models

PAYMENT_METHOD_CHOICES = [
    ("cod", "Cash on delivery"),
    ("eft", "Bank transfer (EFT)"),
    ("card", "Card / online"),
]


class Order(models.Model):

    STATUS_CHOICES = [
        ("pending", "Pending"),
        ("processing", "Processing"),
        ("completed", "Completed"),
        ("cancelled", "Cancelled"),
    ]

    PAYMENT_STATUS_CHOICES = [
        ("unpaid", "Unpaid"),
        ("paid", "Paid"),
    ]

    user = models.ForeignKey(
        settings.AUTH_USER_MODEL,
        on_delete=models.PROTECT,
        related_name="orders",
    )

    total = models.DecimalField(max_digits=10, decimal_places=2)

    status = models.CharField(max_length=20, choices=STATUS_CHOICES, default="pending")

    payment_method = models.CharField(
        max_length=20, choices=PAYMENT_METHOD_CHOICES, default="cod"
    )
    payment_status = models.CharField(
        max_length=20, choices=PAYMENT_STATUS_CHOICES, default="unpaid"
    )

    # delivery details
    full_name = models.CharField(max_length=150, blank=True, default="")
    phone = models.CharField(max_length=30, blank=True, default="")
    address = models.CharField(max_length=255, blank=True, default="")
    city = models.CharField(max_length=100, blank=True, default="")
    postal_code = models.CharField(max_length=20, blank=True, default="")
    notes = models.TextField(blank=True, default="")

    created_at = models.DateTimeField(auto_now_add=True)
    updated_at = models.DateTimeField(auto_now=True)

    @property
    def reference(self):
        """Shown to the customer and used as the EFT payment reference."""
        return f"ORD-{self.pk:05d}"

    def __str__(self):
        return f"Order #{self.pk} - {self.status}"


class OrderItem(models.Model):

    order = models.ForeignKey(
        "orders.Order", on_delete=models.CASCADE, related_name="order_items"
    )
    product = models.ForeignKey(
        "products.Product", on_delete=models.PROTECT, related_name="order_items"
    )
    quantity = models.PositiveIntegerField()
    price = models.DecimalField(max_digits=10, decimal_places=2)

    created_at = models.DateTimeField(auto_now_add=True)
    updated_at = models.DateTimeField(auto_now=True)

    class Meta:
        constraints = [
            models.CheckConstraint(
                condition=models.Q(quantity__gte=1),
                name="order_item_quantity_at_least_one",
            ),
        ]

    def __str__(self):
        return f"{self.product.product_name} x {self.quantity}"

class PaymentMethod(models.Model):
    """Which payment options customers see at checkout, and what we tell
    them after they order. Turn a method on/off with is_enabled."""

    code = models.CharField(max_length=20, unique=True, choices=PAYMENT_METHOD_CHOICES)
    label = models.CharField(max_length=100)
    description = models.CharField(
        max_length=255, blank=True, default="", help_text="Short text shown at checkout."
    )
    instructions = models.TextField(
        blank=True,
        default="",
        help_text="Shown to the customer on their order page after ordering.",
    )
    is_enabled = models.BooleanField(
        default=True, help_text="Untick to show it as 'Coming soon' and refuse orders."
    )
    sort_order = models.PositiveSmallIntegerField(default=0)

    class Meta:
        ordering = ["sort_order", "id"]

    def __str__(self):
        return self.label


class BankAccount(models.Model):
    """Bank details shown to customers who choose EFT. Only ONE account is
    active at a time: activating one switches the others off."""

    ACCOUNT_TYPES = [
        ("business", "Business account"),
        ("cheque", "Cheque / current"),
        ("savings", "Savings"),
        ("transmission", "Transmission"),
    ]

    bank_name = models.CharField(max_length=100)
    account_holder = models.CharField(max_length=150)
    account_number = models.CharField(max_length=30)
    branch_code = models.CharField(max_length=20)
    branch_name = models.CharField(max_length=100, blank=True, default="")
    account_type = models.CharField(max_length=20, choices=ACCOUNT_TYPES, default="business")
    swift_code = models.CharField(
        max_length=20,
        blank=True,
        default="",
        help_text="Only needed if you accept international payments.",
    )
    proof_of_payment_email = models.EmailField(
        blank=True,
        default="",
        help_text="Where customers should send proof of payment.",
    )
    is_active = models.BooleanField(default=True)

    created_at = models.DateTimeField(auto_now_add=True)
    updated_at = models.DateTimeField(auto_now=True)

    class Meta:
        ordering = ["-is_active", "-updated_at"]

    def save(self, *args, **kwargs):
        super().save(*args, **kwargs)
        if self.is_active:
            BankAccount.objects.exclude(pk=self.pk).filter(is_active=True).update(is_active=False)

    def __str__(self):
        return f"{self.bank_name} - {self.account_holder}"


class PaymentGateway(models.Model):
    """NON-SECRET settings for an online card gateway.

    SECRET values (merchant key, secret key, passphrase) must NOT be stored
    here. Put them in environment variables and read them in settings.py.
    """

    PROVIDERS = [
        ("payfast", "PayFast"),
        ("yoco", "Yoco"),
        ("ozow", "Ozow"),
        ("stripe", "Stripe"),
        ("paypal", "PayPal"),
        ("other", "Other"),
    ]

    provider = models.CharField(max_length=20, choices=PROVIDERS)
    is_sandbox = models.BooleanField(
        default=True, help_text="Test mode. Untick only when you go live."
    )
    is_active = models.BooleanField(default=False)
    merchant_id = models.CharField(
        max_length=100, blank=True, default="", help_text="Public merchant identifier."
    )
    public_key = models.CharField(
        max_length=255, blank=True, default="", help_text="Publishable key, if the provider has one."
    )
    notes = models.TextField(
        blank=True, default="", help_text="Reminders for yourself. Never paste secret keys here."
    )

    created_at = models.DateTimeField(auto_now_add=True)
    updated_at = models.DateTimeField(auto_now=True)

    def __str__(self):
        mode = "sandbox" if self.is_sandbox else "live"
        return f"{self.get_provider_display()} ({mode})"


DEFAULT_PAYMENT_METHODS = [
    # code, label, checkout description, after-order instructions, enabled, order
    ("cod", "Cash on delivery", "Pay in cash when your order arrives.",
     "Please have the exact amount ready when your order arrives.", True, 1),
    ("eft", "Bank transfer (EFT)",
     "Pay by EFT. We'll show our bank details after you order and process it once payment reflects.",
     "Use your order reference as the payment reference. We process your order once the payment reflects.",
     True, 2),
    ("card", "Card / online payment", "Pay securely online with your card.", "", False, 3),
]


def ensure_default_payment_methods():
    """Creates the three standard methods the first time they are needed."""
    if PaymentMethod.objects.exists():
        return
    for code, label, description, instructions, enabled, order in DEFAULT_PAYMENT_METHODS:
        PaymentMethod.objects.get_or_create(
            code=code,
            defaults={
                "label": label,
                "description": description,
                "instructions": instructions,
                "is_enabled": enabled,
                "sort_order": order,
            },
        )

from django.contrib import admin, messages
from django.db import transaction
from django.db.models import F
from django.utils import timezone

from products.models import Product

from .models import BankAccount, Order, OrderItem, PaymentGateway, PaymentMethod


class OrderItemInline(admin.TabularInline):
    model = OrderItem
    extra = 0
    readonly_fields = ("created_at", "updated_at")


@admin.register(Order)
class OrderAdmin(admin.ModelAdmin):
    list_display = (
        "id", "full_name", "phone", "total",
        "payment_method", "payment_status", "status", "created_at",
    )
    list_display_links = ("id", "full_name")
    list_editable = ("status", "payment_status")   # change these right in the list
    list_filter = ("status", "payment_status", "payment_method", "created_at")
    search_fields = ("user__email", "full_name", "phone", "address")
    inlines = [OrderItemInline]
    actions = ["mark_processing", "mark_completed", "mark_paid", "cancel_and_restock"]

    @admin.action(description="Mark selected orders as Processing")
    def mark_processing(self, request, queryset):
        n = queryset.exclude(status="cancelled").update(
            status="processing", updated_at=timezone.now()
        )
        self.message_user(request, f"{n} order(s) marked as processing.")

    @admin.action(description="Mark selected orders as Completed")
    def mark_completed(self, request, queryset):
        n = queryset.exclude(status="cancelled").update(
            status="completed", updated_at=timezone.now()
        )
        self.message_user(request, f"{n} order(s) marked as completed.")

    @admin.action(description="Mark selected orders as Paid")
    def mark_paid(self, request, queryset):
        n = queryset.update(payment_status="paid", updated_at=timezone.now())
        self.message_user(request, f"{n} order(s) marked as paid.")

    @admin.action(description="Cancel selected orders and return items to stock")
    @transaction.atomic
    def cancel_and_restock(self, request, queryset):
        count = 0
        orders = queryset.exclude(status="cancelled").prefetch_related("order_items")
        for order in orders:
            for item in order.order_items.all():
                Product.objects.filter(pk=item.product_id).update(
                    current_stock=F("current_stock") + item.quantity
                )
            order.status = "cancelled"
            order.save(update_fields=["status", "updated_at"])
            count += 1
        self.message_user(
            request, f"{count} order(s) cancelled and stock restored.", messages.SUCCESS
        )
